\begin{frame}{Adapter: nối biểu diễn nguồn với decoder}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.53\textwidth}
      \begin{block}{Ý tưởng}
        \begin{itemize}
          \item Lấy hidden states từ LLM đọc nguồn.
          \item Trộn thông tin từ nhiều layer của LLM.
          \item Gated projection đưa biểu diễn sang không gian decoder.
          \item Decoder đọc source memory bằng cross-attention.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{center}
      \begin{tikzpicture}[node distance=0.5cm, every node/.style={font=\small}]
        \node[draw,rounded corners,fill=LLMBlue,minimum width=3.2cm,minimum height=0.7cm] (enc) {Hidden states};
        \node[draw,rounded corners,fill=LLMGreen,below=of enc,minimum width=3.2cm,minimum height=0.7cm] (fuse) {Layer fusion};
        \node[draw,rounded corners,fill=LLMOrange,below=of fuse,minimum width=3.2cm,minimum height=0.7cm] (gate) {Gated projection};
        \node[draw,rounded corners,fill=LLMBlue,below=of gate,minimum width=3.2cm,minimum height=0.7cm] (mem) {Source memory};
        \draw[-{Latex[length=2mm]},thick] (enc) -- (fuse);
        \draw[-{Latex[length=2mm]},thick] (fuse) -- (gate);
        \draw[-{Latex[length=2mm]},thick] (gate) -- (mem);
      \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}

  \vfill
  \centering
  \normalsize\alert{Mục tiêu: dùng lại biểu diễn của LLM, nhưng vẫn giữ cấu trúc encoder-decoder.}

  \smallskip
  \small Adapter không tạo encoder mới từ đầu. Nó chuyển hidden states của LLM thành source memory cho decoder.
\end{frame}

\begin{frame}{Thông điệp chính}

    \item LLM2Seq dùng LLM để đọc nguồn, rồi chuyển source memory cho decoder.
    \item Qwen tốt hơn Llama trên WikiLingua và LR-Sum tiếng Việt.

  \end{itemize}
