\begin{frame}{Memory Adapter: từ LLM hidden states đến decoder memory}
  \vspace{-0.85em}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \begin{block}{Kiến trúc chi tiết adapter}
        \small
        \begin{enumerate}
          \setlength{\itemsep}{0pt}
          \setlength{\parsep}{0pt}
          \item \textbf{Token-wise Layer Fusion}: mỗi token có thể cần thông tin từ độ sâu LLM khác nhau.
          \item \textbf{Gated Residual Projection}: chiếu sang decoder space qua residual path để hạn chế mất thông tin nguồn.
          \item \textbf{Token Salience Gate}: nhấn mạnh token quan trọng mà không loại token khác.
          \item \textbf{EncStack}: cho source token đã thích nghi tương tác và tinh chỉnh lẫn nhau.
          \item \textbf{Global Memory Tokens}: cung cấp document-level workspace cho decoder.
        \end{enumerate}
      \end{block}
    \end{column}
    \begin{column}{0.44\textwidth}

        \begin{minipage}[t][2.1cm][t]{\linewidth}

          \setlength{\itemsep}{0.1em}

\begin{frame}{Backup: phát biểu bài toán}
  \begin{block}{Mục tiêu}
    Tìm mô hình ước lượng phân phối của bản tóm tắt $y = (y_1, \ldots, y_n)$ dựa trên văn bản nguồn $x = (x_1, \ldots, x_m)$:
    \[p(y|x)=\prod_{t=1}^{n}p(y_t \mid y_{<t},x)\]
  \end{block}
  \begin{block}{Công thức hóa LLM2Seq}
    \begin{itemize}
      \item \textbf{Encoder}: $H=E_{\theta}(x)$; \textbf{Adapter}: $M=A_{\phi}(H)$.
      \item \textbf{Decoder}: $p(y|x)=\prod_{t=1}^{n}p_{\psi}(y_t \mid y_{<t},M)$.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Backup: khi nào MTP thực sự nhanh hơn?}
  \begin{block}{Chi phí gần đúng}
    \[C_{AR}\approx nC_{AR\text{-}step}\qquad C_{MTP}\approx\frac{n}{\bar e}(C_{AR\text{-}step}+C_{ver})\]
  \end{block}
  \vfill
  \begin{center}
    \Large $\bar e > 1 + \dfrac{C_{ver}}{C_{AR\text{-}step}}$
  \end{center}
  \vfill
  \small\alert{Số token chấp nhận trung bình phải đủ lớn để bù chi phí verification.}
\end{frame}
